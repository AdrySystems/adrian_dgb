Código VBA Simulado

Sub GenerarReporteMensual()
    Sheets("Reporte").Select
    Range("A1:E1").Font.Bold = True
    Range("A1:E1").Interior.Color = RGB(16, 185, 129)
    MsgBox "¡Reporte generado con éxito por la Macro!", vbInformation
End Sub
